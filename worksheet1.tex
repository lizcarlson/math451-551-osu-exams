\section*{Exam Problem: FLOP Counting Using Nested Loops}

\textbf{Problem (10 points).}

Consider the following algorithm for multiplying two matrices
$A$ and $B$, where both matrices are $n\times n$.

\begin{algorithm}
\caption{Matrix-Matrix Multiplication}
\begin{algorithmic}[1]
\For{$i=1,\ldots,n$}
    \For{$j=1,\ldots,n$}
        \State $C[i,j] = 0$
        \For{$k=1,\ldots,n$}
            \State $C[i,j] = C[i,j] + A[i,k]B[k,j]$
        \EndFor
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

Answer the following questions:

\begin{enumerate}
    \item (2 points) How many times is the innermost loop executed?

    \item (2 points) How many floating-point multiplications
    are performed?

    \item (2 points) How many floating-point additions
    are performed?

    \item (2 points) What is the total number of FLOPs
    as a function of $n$?

    \item (2 points) What is the computational complexity
    (Big-O notation) of this algorithm?
\end{enumerate}

\textbf{Assumptions:}
\begin{itemize}
    \item Each multiplication counts as one FLOP.
    \item Each addition counts as one FLOP.
    \item Loop indexing, comparisons, and assignments
    are not counted as FLOPs.
    \item Initializing $C[i,j]=0$ is not counted as a FLOP.
\end{itemize}

\newpage

\section*{Instructor's Solution}

\begin{enumerate}

    \item \textbf{Number of innermost loop executions}

    There are three nested loops, each running $n$ times.
    Therefore,
    \[
    \boxed{n^3}
    \]

    \item \textbf{Number of multiplications}

    Each iteration of the innermost loop performs one
    multiplication. Therefore,
    \[
    \boxed{n^3}
    \]

    \item \textbf{Number of additions}

    Each iteration of the innermost loop performs one
    addition. Therefore,
    \[
    \boxed{n^3}
    \]

    \item \textbf{Total number of FLOPs}

    Adding the multiplications and additions:
    \[
    \begin{aligned}
    \text{FLOPs}
    &= n^3+n^3\\
    &= \boxed{2n^3}.
    \end{aligned}
    \]

    \item \textbf{Computational complexity}

    Ignoring constant factors, the computational complexity is
    \[
    \boxed{O(n^3)}.
    \]

\end{enumerate}

\subsection*{Optional Numerical Question}

Suppose $n=100$. How many FLOPs does the algorithm require?

\[
\begin{aligned}
\text{FLOPs}
&=2(100)^3\\
&=2,000,000.
\end{aligned}
\]

Thus,
\[
\boxed{2\times 10^6\text{ FLOPs}}
\]
